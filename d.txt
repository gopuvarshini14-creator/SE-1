
# STEP 1: Create folder
mkdir C:\DockerProjects\MyNginx
cd C:\DockerProjects\MyNginx

# STEP 2: Create HTML file
notepad index.html

# Paste and save:
# <html>
# <body>
# <h1>Welcome to My Customized Docker Image!</h1>
# <p>Created by Varshini</p>
# </body>
# </html>

# STEP 3: Create Dockerfile
notepad Dockerfile

# Paste and save:
# FROM nginx:latest
# COPY index.html /usr/share/nginx/html/index.html
# EXPOSE 80
# CMD ["nginx", "-g", "daemon off;"]

# STEP 4: Build customized image
docker build -t mynginx .

# STEP 5: Check images
docker images

# STEP 6: Run container
docker run -d --name mycontainer -p 8080:80 mynginx

# STEP 7: Check running containers
docker ps

# STEP 8: Test in browser
# http://localhost:8080

# STEP 9: Login to Docker Hub
docker login

# STEP 10: Tag image
docker tag mynginx YOUR_USERNAME/mynginx:latest

# STEP 11: Push image to Docker Hub
docker push YOUR_USERNAME/mynginx:latest
